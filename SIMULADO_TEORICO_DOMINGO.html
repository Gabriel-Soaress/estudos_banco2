CREATE OR REPLACE FUNCTION FNC_CLASSIFICA(p_alvo NUMBER, p_rec NUMBER) RETURN VARCHAR2 IS
    v_msg VARCHAR2(50);
BEGIN
    CASE
        WHEN p_rec >= p_alvo THEN v_msg := 'SUPERADA';
        WHEN p_rec < p_alvo  THEN v_msg := 'EM EVOLUCAO';
        ELSE v_msg := 'SEM RECORDE';
    END CASE;
    RETURN v_msg;
END;
/

CREATE OR REPLACE VIEW VW_TESTE_METAS AS
SELECT m.COD_META, NVL(res.MAIOR_REC, 0) AS REC,
       FNC_CLASSIFICA(m.CARGA_ALVO, NVL(res.MAIOR_REC, 0)) AS SITUACAO

Questão 8 • Caça aos Bugs: Questão 4 do Professor (Maior e Menor Nota por Sexo)

Para responder à questão do professor "Homens com maior e menor nota; Mulheres idem" em uma única consulta lado a lado por SEXO, um aluno montou as subqueries no FROM assim: 

SELECT
    resumo.SEXO,
    maior.NOME AS ALUNO_MAIOR_NOTA,
    resumo.MAIOR_NOTA,
    menor.NOME AS ALUNO_MENOR_NOTA,
    resumo.MENOR_NOTA
FROM (
    SELECT a.SEXO, MAX(n.VALOR_NOTA) AS MAIOR_NOTA, MIN(n.VALOR_NOTA) AS MENOR_NOTA
    FROM ALUNO a
    INNER JOIN NOTA n ON n.ID_ALUNO = a.ID_ALUNO
    GROUP BY a.SEXO
) resumo
INNER JOIN (
    SELECT a.NOME, a.SEXO, n.VALOR_NOTA
    FROM ALUNO a INNER JOIN NOTA n ON n.ID_ALUNO = a.ID_ALUNO
) maior ON maior.VALOR_NOTA = resumo.MAIOR_NOTA
INNER JOIN (
    SELECT a.NOME, a.SEXO, n.VALOR_NOTA
    FROM ALUNO a INNER JOIN NOTA n ON n.ID_ALUNO = a.ID_ALUNO
) menor ON menor.VALOR_NOTA = resumo.MENOR_NOTA;

As 3 subqueries estão bem montadas, mas faltou uma condição crucial nos dois ON lá de fora! Se a maior nota entre as Mulheres for 9.5, e um Homem também tiver tirado 9.5 (mesmo que a maior nota dos Homens seja 10.0), o nome desse Homem vai aparecer na linha das Mulheres! O que faltou comparar nos dois ON para não misturar Homens com Mulheres?
